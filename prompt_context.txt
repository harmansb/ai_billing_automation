Input: Compared to March month which projects costs has increased in April month?
Output:
WITH MarchCosts AS (
    SELECT project.id AS project_id, SUM(cost) AS march_cost
    FROM `testproject-198108.all_billing_data.gcp_billing_export_resource_v1_0187DD_4F40BC_35EAB2`
    WHERE EXTRACT(MONTH FROM usage_start_time) = 3
      AND EXTRACT(YEAR FROM usage_start_time) = EXTRACT(YEAR FROM CURRENT_DATE())
    GROUP BY project_id
),
AprilCosts AS (
    SELECT project.id AS project_id, SUM(cost) AS april_cost
    FROM `testproject-198108.all_billing_data.gcp_billing_export_resource_v1_0187DD_4F40BC_35EAB2`
    WHERE EXTRACT(MONTH FROM usage_start_time) = 4
      AND EXTRACT(YEAR FROM usage_start_time) = EXTRACT(YEAR FROM CURRENT_DATE())
    GROUP BY project_id
)
SELECT ac.project_id
FROM AprilCosts ac
JOIN MarchCosts mc ON ac.project_id = mc.project_id
WHERE ac.april_cost > mc.march_cost;
