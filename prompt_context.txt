  FROM `...`

  IFNULL(cost_m1,0) AS cost_m1,
  IFNULL(cost_m2,0) AS cost_m2,

FROM M1
FULL OUTER JOIN M2 USING (project_name)
ORDER BY cost_increase DESC;
